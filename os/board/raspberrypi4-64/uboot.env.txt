# ─── Domus OS - U-Boot Default Environment Variables ─────────────────────────
# Initial configuration for dual-slot A/B bootcounter and RAUC state.
boot_slot=a
rauc_slot=a
bootcount=0
bootlimit=3
upgrade_available=0
autoload=no
baudrate=115200
bootcmd=run boot_rauc
boot_rauc=test -n "${boot_slot}" || setenv boot_slot a; test -n "${bootcount}" || setenv bootcount 0; test -n "${bootlimit}" || setenv bootlimit 3; echo "Domus OS active slot: ${boot_slot}"; if test "${bootcount}" -ge "${bootlimit}"; then echo "Bootlimit exceeded! Rolling back slot..."; if test "${boot_slot}" = "a"; then setenv boot_slot b; else setenv boot_slot a; fi; setenv bootcount 0; saveenv; fi; setexpr bootcount ${bootcount} + 1; saveenv; if test "${boot_slot}" = "a"; then setenv rootpart domus-root-a; else setenv rootpart domus-root-b; fi; setenv bootargs "console=serial0,115200 console=tty1 root=/dev/disk/by-partlabel/${rootpart} rootfstype=squashfs ro rootwait quiet cgroup_enable=cpuset cgroup_memory=1 cgroup_enable=memory"; load mmc 0:1 ${kernel_addr_r} Image; load mmc 0:1 ${fdt_addr_r} bcm2711-rpi-4-b.dtb; booti ${kernel_addr_r} - ${fdt_addr_r}
